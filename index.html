const SS = SpreadsheetApp.getActiveSpreadsheet(); const HISTORY_SHEET_NAME = "Scout_Data"; const CURRENT_SHEET_NAME = "Current_Week"; const HISTORY_COLUMNS = 9; const CURRENT_COLUMNS = 11; const STATUS_CACHE_KEY = "ipm_current_week_status"; const ENTRIES_CACHE_KEY = "ipm_current_week_entries"; /* ================================================== WEB APP ROUTES ================================================== */ function doGet(e) { try { const parameters = e && e.parameter ? e.parameter : {}; const action = parameters.action || ""; const callback = parameters.callback || ""; let result = { success: true, message: "IPM Scout API is working" }; if (action === "getStatus") { result = getStatus(); } else if (action === "getEntry") { result = getEntry( parameters.section || "", parameters.bay || "" ); } else if (action === "getCurrentWeekEntries") { result = getCurrentWeekEntries(); } else if (action === "getDashboard") { result = getDashboard(); } else if (action === "getHeatMaps") { result = getHeatMaps(); } else if (action === "getAppData") { result = getAppData(); } else if (action === "rebuildCurrentWeek") { result = rebuildCurrentWeekFromScoutData(); } return createGetResponse(result, callback); } catch (error) { const callback = e && e.parameter ? e.parameter.callback || "" : ""; return createGetResponse( { success: false, message: error.message }, callback ); } } /* ================================================== SAVE DATA Scout_Data remains the permanent history. Current_Week keeps one latest row per bay. ================================================== */ function doPost(e) { const lock = LockService.getScriptLock(); try { if (!lock.tryLock(10000)) { throw new Error( "Another scout is saving data. Please try again." ); } const parameters = e && e.parameter ? e.parameter : {}; const now = new Date(); const week = getWeekNumber(now); const scout = String( parameters.scout || "Raissa" ).trim(); const section = normalizeSection( parameters.section || "" ); const bay = String( parameters.bay || "" ).trim(); const status = String( parameters.status || "Scouted" ).trim(); const notes = String( parameters.notes || "" ).trim(); if (!section) { throw new Error("Section is required."); } if (!bay) { throw new Error("Bay is required."); } const values = { thrips: toNumber(parameters.thrips), aphids: toNumber(parameters.aphids), whiteflies: toNumber(parameters.whiteflies), fungusGnats: toNumber(parameters.fungusGnats) }; /* * 1. Append to permanent history. * Previous submissions are not deleted. */ appendHistoryRows({ now: now, week: week, scout: scout, section: section, bay: bay, status: status, notes: notes, values: values }); /* * 2. Update one current row for this bay. */ upsertCurrentWeekRow({ now: now, week: week, scout: scout, section: section, bay: bay, status: status, notes: notes, values: values }); clearCurrentWeekCaches(); return createPostResponse({ success: true, message: status === "Empty" ? "Empty bay updated" : "Counts updated" }); } catch (error) { return createPostResponse({ success: false, message: error.message }); } finally { try { lock.releaseLock(); } catch (error) { // Lock was not acquired or was already released. } } } /* ================================================== RESPONSE HELPERS ================================================== */ function createGetResponse(result, callback) { if (callback) { return ContentService .createTextOutput( callback + "(" + JSON.stringify(result) + ")" ) .setMimeType( ContentService.MimeType.JAVASCRIPT ); } return ContentService .createTextOutput( JSON.stringify(result) ) .setMimeType( ContentService.MimeType.JSON ); } function createPostResponse(result) { return ContentService .createTextOutput( JSON.stringify(result) ) .setMimeType( ContentService.MimeType.JSON ); } /* ================================================== SHEET HELPERS ================================================== */ function getHistorySheet() { const sheet = SS.getSheetByName(HISTORY_SHEET_NAME); if (!sheet) { throw new Error( 'Sheet "' + HISTORY_SHEET_NAME + '" was not found.' ); } return sheet; } function getCurrentWeekSheet() { let sheet = SS.getSheetByName(CURRENT_SHEET_NAME); if (!sheet) { sheet = SS.insertSheet(CURRENT_SHEET_NAME); } ensureCurrentWeekHeaders(sheet); return sheet; } function ensureCurrentWeekHeaders(sheet) { const headers = [[ "Week", "Section", "Bay", "Status", "Thrips", "Aphids", "Whiteflies", "FG", "Scout", "Timestamp", "Notes" ]]; const existing = sheet .getRange(1, 1, 1, CURRENT_COLUMNS) .getValues()[0]; let needsHeaders = false; for (let i = 0; i < headers[0].length; i++) { if (String(existing[i] || "").trim() !== headers[0][i]) { needsHeaders = true; break; } } if (needsHeaders) { sheet .getRange(1, 1, 1, CURRENT_COLUMNS) .setValues(headers); sheet.setFrozenRows(1); } } function getHistoryData() { const sheet = getHistorySheet(); const lastRow = sheet.getLastRow(); if (lastRow < 2) { return []; } return sheet .getRange( 2, 1, lastRow - 1, HISTORY_COLUMNS ) .getValues(); } function getCurrentWeekData() { const sheet = ensureCurrentWeekReady(); const lastRow = sheet.getLastRow(); if (lastRow < 2) { return []; } return sheet .getRange( 2, 1, lastRow - 1, CURRENT_COLUMNS ) .getValues(); } /* ================================================== CURRENT WEEK INITIALIZATION / WEEKLY RESET If Current_Week is empty or contains an older week, rebuild it from Scout_Data for the present week. ================================================== */ function ensureCurrentWeekReady() { const sheet = getCurrentWeekSheet(); const currentWeek = getWeekNumber(new Date()); const lastRow = sheet.getLastRow(); if (lastRow < 2) { rebuildCurrentWeekFromScoutData(); return sheet; } const storedWeek = Number( sheet.getRange(2, 1).getValue() || 0 ); if (storedWeek !== currentWeek) { rebuildCurrentWeekFromScoutData(); } return sheet; } function rebuildCurrentWeekFromScoutData() { const currentSheet = getCurrentWeekSheet(); const currentWeek = getWeekNumber(new Date()); const historyData = getHistoryData(); /* * Keep only the latest submission for each section/bay. * Scout_Data is ordered chronologically, so later rows * overwrite earlier values in the map. */ const entries = {}; for (let i = 0; i < historyData.length; i++) { const row = historyData[i]; const timestamp = row[0]; const week = Number(row[1] || 0); const scout = String(row[2] || "").trim(); const section = normalizeSection(row[3]); const bay = String(row[4] || "").trim(); const status = String(row[5] || "").trim(); const pest = String(row[6] || "").trim(); const count = toNumber(row[7]); const notes = String(row[8] || "").trim(); if ( week !== currentWeek || !section || !bay ) { continue; } const key = section + "|" + bay; if (!entries[key]) { entries[key] = { week: currentWeek, section: section, bay: bay, status: status, thrips: "", aphids: "", whiteflies: "", fungusGnats: "", scout: scout, timestamp: timestamp, notes: notes }; } entries[key].status = status; entries[key].scout = scout; entries[key].timestamp = timestamp; if (notes) { entries[key].notes = notes; } if (status === "Empty") { entries[key].thrips = ""; entries[key].aphids = ""; entries[key].whiteflies = ""; entries[key].fungusGnats = ""; continue; } if (pest === "Thrips") { entries[key].thrips = count; } if (pest === "Aphids") { entries[key].aphids = count; } if (pest === "Whiteflies") { entries[key].whiteflies = count; } if (pest === "Fungus Gnats") { entries[key].fungusGnats = count; } } /* * Clear only data rows. Headers and formatting remain. */ const currentLastRow = currentSheet.getLastRow(); if (currentLastRow > 1) { currentSheet .getRange( 2, 1, currentLastRow - 1, CURRENT_COLUMNS ) .clearContent(); } const rows = Object.keys(entries) .sort() .map(key => { const item = entries[key]; return [ item.week, item.section, item.bay, item.status, item.thrips, item.aphids, item.whiteflies, item.fungusGnats, item.scout, item.timestamp, item.notes ]; }); if (rows.length > 0) { currentSheet .getRange( 2, 1, rows.length, CURRENT_COLUMNS ) .setValues(rows); } clearCurrentWeekCaches(); return { success: true, week: currentWeek, rows: rows.length, message: "Current_Week rebuilt successfully" }; } /* ================================================== HISTORY WRITE ================================================== */ function appendHistoryRows(entry) { const sheet = getHistorySheet(); let rows = []; if (entry.status === "Empty") { rows = [[ entry.now, entry.week, entry.scout, entry.section, entry.bay, "Empty", "", "", entry.notes || "No plants" ]]; } else { rows = [ [ entry.now, entry.week, entry.scout, entry.section, entry.bay, "Scouted", "Thrips", entry.values.thrips, entry.notes ], [ entry.now, entry.week, entry.scout, entry.section, entry.bay, "Scouted", "Aphids", entry.values.aphids, entry.notes ], [ entry.now, entry.week, entry.scout, entry.section, entry.bay, "Scouted", "Whiteflies", entry.values.whiteflies, entry.notes ], [ entry.now, entry.week, entry.scout, entry.section, entry.bay, "Scouted", "Fungus Gnats", entry.values.fungusGnats, entry.notes ] ]; } sheet .getRange( sheet.getLastRow() + 1, 1, rows.length, HISTORY_COLUMNS ) .setValues(rows); } /* ================================================== CURRENT WEEK UPSERT ================================================== */ function upsertCurrentWeekRow(entry) { const sheet = ensureCurrentWeekReady(); const lastRow = sheet.getLastRow(); let targetRow = -1; if (lastRow >= 2) { const keys = sheet .getRange( 2, 2, lastRow - 1, 2 ) .getValues(); for (let i = 0; i < keys.length; i++) { const rowSection = normalizeSection(keys[i][0]); const rowBay = String(keys[i][1] || "").trim(); if ( rowSection === entry.section && rowBay === entry.bay ) { targetRow = i + 2; break; } } } const row = [[ entry.week, entry.section, entry.bay, entry.status, entry.status === "Empty" ? "" : entry.values.thrips, entry.status === "Empty" ? "" : entry.values.aphids, entry.status === "Empty" ? "" : entry.values.whiteflies, entry.status === "Empty" ? "" : entry.values.fungusGnats, entry.scout, entry.now, entry.notes || ( entry.status === "Empty" ? "No plants" : "" ) ]]; if (targetRow === -1) { targetRow = sheet.getLastRow() + 1; } sheet .getRange( targetRow, 1, 1, CURRENT_COLUMNS ) .setValues(row); } /* ================================================== GENERAL HELPERS ================================================== */ function normalizeSection(section) { const value = String(section || "").trim(); const sectionMap = { "9A": "S9A", "9B": "S9B", "10A": "S10A", "10B": "S10B" }; return sectionMap[value] || value; } function toNumber(value) { const number = Number(value || 0); return Number.isFinite(number) ? number : 0; } function clearCurrentWeekCaches() { const cache = CacheService.getScriptCache(); cache.remove(STATUS_CACHE_KEY); cache.remove(ENTRIES_CACHE_KEY); } /* ================================================== WEEK NUMBER ================================================== */ function getWeekNumber(date) { const d = new Date( Date.UTC( date.getFullYear(), date.getMonth(), date.getDate() ) ); const dayNumber = d.getUTCDay() || 7; d.setUTCDate( d.getUTCDate() + 4 - dayNumber ); const yearStart = new Date( Date.UTC( d.getUTCFullYear(), 0, 1 ) ); return Math.ceil( ( ( (d - yearStart) / 86400000 ) + 1 ) / 7 ); } /* ================================================== CURRENT WEEK STATUS ================================================== */ function getStatus(sharedData) { const cache = CacheService.getScriptCache(); const cached = cache.get(STATUS_CACHE_KEY); if (cached) { return JSON.parse(cached); } const data = sharedData || getCurrentWeekData(); const currentWeek = getWeekNumber(new Date()); const records = {}; for (let i = 0; i < data.length; i++) { const row = data[i]; const section = normalizeSection(row[1]); const bay = String(row[2] || "").trim(); const status = String(row[3] || "").trim(); if (section && bay) { records[section + "|" + bay] = status; } } const result = { success: true, week: currentWeek, records: records }; cache.put( STATUS_CACHE_KEY, JSON.stringify(result), 30 ); return result; } /* ================================================== ALL CURRENT-WEEK ENTRIES ================================================== */ function getCurrentWeekEntries(sharedData) { const cache = CacheService.getScriptCache(); const cached = cache.get(ENTRIES_CACHE_KEY); if (cached) { return JSON.parse(cached); } const data = sharedData || getCurrentWeekData(); const currentWeek = getWeekNumber(new Date()); const entries = {}; for (let i = 0; i < data.length; i++) { const row = data[i]; const section = normalizeSection(row[1]); const bay = String(row[2] || "").trim(); if (!section || !bay) { continue; } const key = section + "|" + bay; entries[key] = { section: section, bay: bay, status: String(row[3] || "").trim(), counts: { "Thrips": row[4] === "" ? "" : toNumber(row[4]), "Aphids": row[5] === "" ? "" : toNumber(row[5]), "Whiteflies": row[6] === "" ? "" : toNumber(row[6]), "Fungus Gnats": row[7] === "" ? "" : toNumber(row[7]) }, lastScout: String(row[8] || "").trim(), lastUpdated: row[9] || "", notes: String(row[10] || "").trim() }; } const result = { success: true, week: currentWeek, entries: entries }; cache.put( ENTRIES_CACHE_KEY, JSON.stringify(result), 30 ); return result; } /* ================================================== SINGLE BAY ENTRY ================================================== */ function getEntry(section, bay) { const normalizedSection = normalizeSection(section); const normalizedBay = String(bay || "").trim(); const currentWeek = getWeekNumber(new Date()); const data = getCurrentWeekData(); const result = { success: true, week: currentWeek, section: normalizedSection, bay: normalizedBay, status: "", counts: { "Thrips": "", "Aphids": "", "Whiteflies": "", "Fungus Gnats": "" }, notes: "", lastUpdated: "", lastScout: "" }; for (let i = 0; i < data.length; i++) { const rowSection = normalizeSection(data[i][1]); const rowBay = String(data[i][2] || "").trim(); if ( rowSection !== normalizedSection || rowBay !== normalizedBay ) { continue; } result.status = String(data[i][3] || "").trim(); result.counts = { "Thrips": data[i][4] === "" ? "" : toNumber(data[i][4]), "Aphids": data[i][5] === "" ? "" : toNumber(data[i][5]), "Whiteflies": data[i][6] === "" ? "" : toNumber(data[i][6]), "Fungus Gnats": data[i][7] === "" ? "" : toNumber(data[i][7]) }; result.lastScout = String(data[i][8] || "").trim(); result.lastUpdated = data[i][9] || ""; result.notes = String(data[i][10] || "").trim(); break; } return result; } /* ================================================== ONE-REQUEST APP DATA Reads Current_Week only once and returns everything needed by Scout, Dashboard, Heat Maps and Reports. ================================================== */ function getAppData() { const data = getCurrentWeekData(); const currentWeek = getWeekNumber(new Date()); const statusResult = getStatus(data); const entriesResult = getCurrentWeekEntries(data); const dashboardResult = getDashboard(data); const heatMapsResult = getHeatMaps(data); return { success: true, week: currentWeek, records: statusResult.records || {}, entries: entriesResult.entries || {}, dashboard: dashboardResult, heatMaps: heatMapsResult }; } /* ================================================== DASHBOARD ================================================== */ function getDashboard(sharedData) { const data = sharedData || getCurrentWeekData(); const currentWeek = getWeekNumber(new Date()); const sectionOrder = [ "S1", "S2", "S3", "S4", "S5", "S6", "S7", "S8", "S9", "S9A", "S9B", "S10A", "S10B" ]; const bayMap = { "S1": ["B1","B2","B3","B4","B5","B6"], "S2": ["B2","B3","B4","B5","B6","B7","B8","B9","B10","B11","B12","B13","B14","B15","B16"], "S3": ["B2","B3","B4","B5","B6","B7","B8","B9","B10","B11","B12","B13","B14","B15","B16"], "S4": ["B1","B2","B3","B4","B5","B6","B7","B8","B9","B10","B11","B12","B13","B14","B15","B16"], "S5": ["B4","B5","B6","B7","B8","B9","B10"], "S6": ["B2","B3","B4","B5","B6","B7","B8","B9","B10"], "S7": ["B2","B3","B4","B5","B6","B7","B8","B9","B10"], "S8": ["B1","B2","B3","B4","B5","B6","B7","B8","B9"], "S9": ["B1","B2","B3","B4","B5"], "S9A": ["B6","B7","B8","B9","B10","B11","B12"], "S9B": ["B6","B7","B8","B9","B10","B11","B12"], "S10A": ["B1","B2","B3","B4","B5","B6","B7","B8","B9","B10"], "S10B": ["B1","B2","B3","B4","B5","B6","B7","B8","B9","B10"] }; const pests = [ "Thrips", "Aphids", "Whiteflies", "Fungus Gnats" ]; const sectionData = {}; sectionOrder.forEach(section => { sectionData[section] = { progress: 0, completed: 0, total: bayMap[section].length, pests: { "Thrips": 0, "Aphids": 0, "Whiteflies": 0, "Fungus Gnats": 0 } }; }); for (let i = 0; i < data.length; i++) { const row = data[i]; const section = normalizeSection(row[1]); if (!sectionData[section]) { continue; } sectionData[section].completed++; sectionData[section].pests["Thrips"] += toNumber(row[4]); sectionData[section].pests["Aphids"] += toNumber(row[5]); sectionData[section].pests["Whiteflies"] += toNumber(row[6]); sectionData[section].pests["Fungus Gnats"] += toNumber(row[7]); } sectionOrder.forEach(section => { const item = sectionData[section]; item.progress = item.total > 0 ? Math.round( (item.completed / item.total) * 100 ) : 0; }); const hotspots = {}; pests.forEach(pest => { let topSection = ""; let topValue = 0; sectionOrder.forEach(section => { const value = sectionData[section].pests[pest]; if (value > topValue) { topValue = value; topSection = section; } }); hotspots[pest] = { section: topSection || "-", value: topValue }; }); const totalBays = sectionOrder.reduce( (sum, section) => sum + bayMap[section].length, 0 ); const completed = data.length; const overallProgress = totalBays > 0 ? Math.round( (completed / totalBays) * 100 ) : 0; return { success: true, week: currentWeek, progress: overallProgress, sectionData: sectionData, hotspots: hotspots }; } /* ================================================== HEAT MAPS ================================================== */ function getHeatMaps(sharedData) { const data = sharedData || getCurrentWeekData(); const currentWeek = getWeekNumber(new Date()); const sectionTotals = {}; const bayCounts = {}; for (let i = 0; i < data.length; i++) { const row = data[i]; const section = normalizeSection(row[1]); const bay = String(row[2] || "").trim(); const status = String(row[3] || "").trim(); if ( !section || !bay || status !== "Scouted" ) { continue; } if (!sectionTotals[section]) { sectionTotals[section] = { "Fungus Gnats": 0 }; } if (!bayCounts[section]) { bayCounts[section] = {}; } bayCounts[section][bay] = { "Thrips": toNumber(row[4]), "Aphids": toNumber(row[5]), "Whiteflies": toNumber(row[6]) }; sectionTotals[section]["Fungus Gnats"] += toNumber(row[7]); } return { success: true, week: currentWeek, fungusGnatsBySection: sectionTotals, bayCounts: bayCounts }; }
